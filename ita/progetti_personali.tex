\section{Progetti personali}

\cvitem{}{
  \textbf{Libreria sensore di luce TSL2591} \hfill \small \url{https://github.com/its-fonsy/tsl2591-avr}
  \vspace{1mm}\newline{}\small
  Libreria per il sensore di luce TSL2591 scritta in C per il microcontrollore ATmega328P.
}

\vspace{2mm}

\cvitem{}{
  \textbf{Pomodoro timer} \hfill \small \url{https://github.com/its-fonsy/avr-pomodoro}
  \vspace{1mm}\newline{}\normalsize
  Timer che integra la tecnica del pomodoro realizzato con Atmega328p e display
  OLED SSD1306. Realizzato senza il supporto di librerie esterne, l'utilizzo
  delle periferiche del microcontrollore (timer e i2c) e il display è stato
  implementato da zero.
}

\vspace{2mm}

\cvitem{}{
  \textbf{Lyrpy} \hfill \small \url{https://github.com/its-fonsy/lyrpy}
  \vspace{1mm}\newline{}\small
  Stampa il testo della canzone in riproduzione evidenziando il verso
  attualmente cantato sul terminale.\newline{} Scritto in Python mediante la
  libreria ncurses.
}

\vspace{2mm}

\cvitem{}{
  \textbf{Fastfingers} \hfill \small \url{https://github.com/its-fonsy/fastfingers}
  \vspace{1mm}\newline{}\small
  Replica del gioco di battitura "10FastFingers" scritto in C per il terminale
  con il supporto della libreria di ncurses.
}
